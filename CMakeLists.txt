cmake_minimum_required(VERSION 3.20)
project(l4lb VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_library(l4lb_config src/config/config.cpp)
target_include_directories(l4lb_config PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_compile_options(l4lb_config PRIVATE -Wall -Wextra -Wpedantic)
add_library(l4lb_net src/net/reactor.cpp)
target_include_directories(l4lb_net PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_compile_options(l4lb_net PRIVATE -Wall -Wextra -Wpedantic)
add_library(l4lb_control src/control/tcp_service.cpp)
target_link_libraries(l4lb_control PUBLIC l4lb_net)
add_executable(l4lb src/cli/main.cpp)
target_link_libraries(l4lb PRIVATE l4lb_config l4lb_control)
include(CTest)
if(BUILD_TESTING)
  add_executable(config_test tests/config_test.cpp)
  target_link_libraries(config_test PRIVATE l4lb_config)
  add_test(NAME config_unit COMMAND config_test)
  add_test(NAME cli_integration COMMAND "${CMAKE_COMMAND}"
    "-DPROGRAM=$<TARGET_FILE:l4lb>" "-DSOURCE=${CMAKE_SOURCE_DIR}"
    "-DTMP=${CMAKE_BINARY_DIR}/test-tmp" -P "${CMAKE_SOURCE_DIR}/tests/cli_test.cmake")
  set_tests_properties(config_unit cli_integration PROPERTIES TIMEOUT 20)
endif()
if(BUILD_TESTING)
  find_package(Threads REQUIRED)
  add_executable(net_unit_test tests/net_unit_test.cpp)
  target_link_libraries(net_unit_test PRIVATE l4lb_net)
  add_executable(tcp_test_server tests/tcp_test_server.cpp)
  target_link_libraries(tcp_test_server PRIVATE l4lb_net l4lb_config)
  add_executable(tcp_integration_test tests/tcp_integration_test.cpp)
  target_link_libraries(tcp_integration_test PRIVATE l4lb_net Threads::Threads)
  add_test(NAME s2_net_unit COMMAND net_unit_test)
  add_test(NAME s2_tcp_integration COMMAND tcp_integration_test
    "$<TARGET_FILE:l4lb>" "$<TARGET_FILE:tcp_test_server>" "${CMAKE_BINARY_DIR}/test-s2")
  set_tests_properties(s2_net_unit PROPERTIES LABELS s2 TIMEOUT 20)
  set_tests_properties(s2_tcp_integration PROPERTIES LABELS s2 TIMEOUT 120)
endif()
if(BUILD_TESTING)
  add_executable(reactor_state_test tests/reactor_state_test.cpp)
  target_include_directories(reactor_state_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")
  target_link_libraries(reactor_state_test PRIVATE Threads::Threads)
  add_test(NAME s2_reactor_state COMMAND reactor_state_test)
  set_tests_properties(s2_reactor_state PROPERTIES LABELS s2 TIMEOUT 20)
endif()
if(BUILD_TESTING)
  add_executable(tcp_echo_backend tests/tcp_echo_backend.cpp)
  target_include_directories(tcp_echo_backend PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")
endif()
