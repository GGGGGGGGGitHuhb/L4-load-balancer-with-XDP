cmake_minimum_required(VERSION 3.20)
project(l4lb VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_library(l4lb_config src/config/config.cpp)
target_include_directories(l4lb_config PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_compile_options(l4lb_config PRIVATE -Wall -Wextra -Wpedantic)
add_library(l4lb_net src/net/reactor.cpp src/net/udp_reactor.cpp)
target_include_directories(l4lb_net PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_compile_options(l4lb_net PRIVATE -Wall -Wextra -Wpedantic)
add_library(l4lb_core src/core/scheduler.cpp)

add_library(l4lb_health src/health/checker.cpp)
target_include_directories(l4lb_health PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_compile_options(l4lb_health PRIVATE -Wall -Wextra -Wpedantic)
add_library(l4lb_metrics src/metrics/metrics.cpp)
target_include_directories(l4lb_metrics PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_compile_options(l4lb_metrics PRIVATE -Wall -Wextra -Wpedantic)
add_library(l4lb_control src/control/metrics_service.cpp src/control/health_selection.cpp src/control/tcp_service.cpp src/control/udp_service.cpp src/control/service.cpp)
target_link_libraries(l4lb_control PUBLIC l4lb_net l4lb_core l4lb_health l4lb_metrics)
add_executable(l4lb src/cli/main.cpp)
target_link_libraries(l4lb PRIVATE l4lb_config l4lb_control)
